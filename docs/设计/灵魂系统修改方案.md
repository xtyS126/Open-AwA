# 灵魂系统修改方案

## 一、现状分析

### 1.1 当前实现
Open-AwA 目前的灵魂/人格系统基于以下组件：
- **工作区人设文件**：`SOUL.md`、`IDENTITY.md` 等存储在 workspace 的 `persona_files_json` 中
- **角色引擎**：`role_engine.py` 负责加载角色配置并注入到 Agent 上下文
- **日记模块**：`diary.py` 中的 `_load_soul_content()` 读取 SOUL.md 作为人格描述

### 1.2 存在的问题
1. **缺少开关控制**：灵魂注入是硬编码的，无法在运行时动态关闭
2. **生命周期管理缺失**：没有记录灵魂注入的状态（是否已注入、何时注入等）
3. **缺少降级机制**：当灵魂文件加载失败时，没有明确的降级策略
4. **与记忆系统耦合不清**：灵魂注入和记忆注入的边界不明确

---

## 二、修改目标

### 2.1 核心需求
- **可随时关闭灵魂注入**：提供运行时开关，支持动态启用/禁用
- **完善状态管理**：记录灵魂注入的生命周期状态
- **保持向后兼容**：默认行为不变，新功能可选启用
- **清晰的边界**：明确灵魂系统与记忆系统的职责边界

### 2.2 参考 Agent Diva 的设计
Agent Diva 的灵魂系统包含：
- `SoulState`：记录 bootstrap 生命周期状态
- `SoulStateStore`：持久化灵魂状态
- 明确的注入时机和状态检查

## 三、设计方案

### 3.1 新增灵魂状态管理器

创建 `backend/core/soul_state.py`，实现灵魂状态管理：

```python
from dataclasses import dataclass, field
from datetime import datetime
from typing import Optional
from pathlib import Path
import json

@dataclass
class SoulState:
    """灵魂状态数据类"""
    bootstrap_seeded_at: Optional[datetime] = None  # 首次注入时间
    bootstrap_completed_at: Optional[datetime] = None  # 注入完成时间
    injection_enabled: bool = True  # 是否启用注入（新增开关）
    last_injection_at: Optional[datetime] = None  # 最近一次注入时间
    injection_count: int = 0  # 累计注入次数

class SoulStateManager:
    """灵魂状态管理器"""
    
    def __init__(self, workspace_id: str, state_dir: Optional[Path] = None):
        self.workspace_id = workspace_id
        self.state_dir = state_dir or Path("./data/soul_states")
        self.state_file = self.state_dir / f"{workspace_id}_soul_state.json"
        self._state: Optional[SoulState] = None
    
    def load_state(self) -> SoulState:
        """加载灵魂状态"""
        if self._state is not None:
            return self._state
        
        if self.state_file.exists():
            try:
                with open(self.state_file, 'r', encoding='utf-8') as f:
                    data = json.load(f)
                    self._state = SoulState(
                        bootstrap_seeded_at=datetime.fromisoformat(data['bootstrap_seeded_at']) if data.get('bootstrap_seeded_at') else None,
                        bootstrap_completed_at=datetime.fromisoformat(data['bootstrap_completed_at']) if data.get('bootstrap_completed_at') else None,
                        injection_enabled=data.get('injection_enabled', True),
                        last_injection_at=datetime.fromisoformat(data['last_injection_at']) if data.get('last_injection_at') else None,
                        injection_count=data.get('injection_count', 0)
                    )
            except Exception:
                self._state = SoulState()
        else:
            self._state = SoulState()
        
        return self._state
    
    def save_state(self) -> None:
        """保存灵魂状态"""
        if self._state is None:
            return
        
        self.state_dir.mkdir(parents=True, exist_ok=True)
        
        data = {
            'bootstrap_seeded_at': self._state.bootstrap_seeded_at.isoformat() if self._state.bootstrap_seeded_at else None,
            'bootstrap_completed_at': self._state.bootstrap_completed_at.isoformat() if self._state.bootstrap_completed_at else None,
            'injection_enabled': self._state.injection_enabled,
            'last_injection_at': self._state.last_injection_at.isoformat() if self._state.last_injection_at else None,
            'injection_count': self._state.injection_count
        }
        
        with open(self.state_file, 'w', encoding='utf-8') as f:
            json.dump(data, f, ensure_ascii=False, indent=2)
    
    def is_injection_enabled(self) -> bool:
        """检查是否启用灵魂注入"""
        state = self.load_state()
        return state.injection_enabled
    
    def set_injection_enabled(self, enabled: bool) -> None:
        """设置灵魂注入开关"""
        state = self.load_state()
        state.injection_enabled = enabled
        self.save_state()
    
    def mark_bootstrap_seeded(self) -> None:
        """标记首次注入"""
        state = self.load_state()
        if state.bootstrap_seeded_at is None:
            state.bootstrap_seeded_at = datetime.utcnow()
            self.save_state()
    
    def mark_injection_completed(self) -> None:
        """标记注入完成"""
        state = self.load_state()
        now = datetime.utcnow()
        
        if state.bootstrap_seeded_at is None:
            state.bootstrap_seeded_at = now
        
        if state.bootstrap_completed_at is None:
            state.bootstrap_completed_at = now
        
        state.last_injection_at = now
        state.injection_count += 1
        self.save_state()
    
    def is_bootstrap_completed(self) -> bool:
        """检查是否已完成首次注入"""
        state = self.load_state()
        return state.bootstrap_completed_at is not None
    
    def reset_state(self) -> None:
        """重置灵魂状态（用于测试或重新初始化）"""
        self._state = SoulState()
        self.save_state()
```

### 3.3 提供 API 控制接口

在 `backend/api/routes/workspace.py` 中添加控制接口：

```python
from pydantic import BaseModel

class SoulInjectionToggle(BaseModel):
    enabled: bool

@router.get("/{workspace_id}/soul/injection-status")
def get_soul_injection_status(workspace_id: str, mgr: WorkspaceManager = Depends(_get_mgr)):
    """获取灵魂注入状态"""
    from core.soul_state import SoulStateManager
    manager = SoulStateManager(workspace_id)
    state = manager.load_state()
    
    return {
        "workspace_id": workspace_id,
        "injection_enabled": state.injection_enabled,
        "bootstrap_seeded_at": state.bootstrap_seeded_at.isoformat() if state.bootstrap_seeded_at else None,
        "bootstrap_completed_at": state.bootstrap_completed_at.isoformat() if state.bootstrap_completed_at else None,
        "last_injection_at": state.last_injection_at.isoformat() if state.last_injection_at else None,
        "injection_count": state.injection_count
    }

@router.put("/{workspace_id}/soul/injection-toggle")
def toggle_soul_injection(workspace_id: str, body: SoulInjectionToggle, mgr: WorkspaceManager = Depends(_get_mgr)):
    """切换灵魂注入开关"""
    from core.soul_state import SoulStateManager
    manager = SoulStateManager(workspace_id)
    manager.set_injection_enabled(body.enabled)
    
    return {
        "workspace_id": workspace_id,
        "injection_enabled": body.enabled,
        "message": f"灵魂注入已{'启用' if body.enabled else '禁用'}"
    }

@router.post("/{workspace_id}/soul/reset")
def reset_soul_state(workspace_id: str, mgr: WorkspaceManager = Depends(_get_mgr)):
    """重置灵魂状态（谨慎使用）"""
    from core.soul_state import SoulStateManager
    manager = SoulStateManager(workspace_id)
    manager.reset_state()
    
    return {
        "workspace_id": workspace_id,
        "message": "灵魂状态已重置"
    }
```

### 3.4 前端集成（可选）

在 `frontend/src/features/settings/` 中添加灵魂注入控制 UI：

```typescript
// SoulInjectionToggle.tsx
import { useState, useEffect } from 'react';
import { workspaceApi } from '@/shared/api/workspaceApi';

export function SoulInjectionToggle({ workspaceId }: { workspaceId: string }) {
  const [enabled, setEnabled] = useState(true);
  const [loading, setLoading] = useState(false);

  useEffect(() => {
    // 加载当前状态
    workspaceApi.getSoulInjectionStatus(workspaceId).then(status => {
      setEnabled(status.injection_enabled);
    });
  }, [workspaceId]);

  const handleToggle = async () => {
    setLoading(true);
    try {
      await workspaceApi.toggleSoulInjection(workspaceId, !enabled);
      setEnabled(!enabled);
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="soul-injection-toggle">
      <label>
        <input
          type="checkbox"
          checked={enabled}
          onChange={handleToggle}
          disabled={loading}
        />
        启用灵魂注入
      </label>
      <p className="description">
        关闭后，Agent 将不会加载 SOUL.md 等人格文件
      </p>
    </div>
  );
}
```

- **无破坏性变更**：现有代码无需修改即可继续工作

## 七、风险与缓解

### 7.1 风险
1. **状态文件丢失**：可能导致灵魂注入状态不一致
2. **并发访问**：多个请求同时修改状态文件可能导致冲突

### 7.2 缓解措施
1. **状态文件备份**：定期备份状态文件
2. **文件锁**：使用文件锁防止并发写入冲突
3. **异常处理**：状态文件读取失败时使用默认值
