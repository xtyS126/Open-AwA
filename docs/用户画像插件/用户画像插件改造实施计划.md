```python
# backend/db/models.py — 新增模型

class ProfileFact(Base):
    """
    用户画像事实，存储从对话和行为中提取的结构化用户信息。
    设计原则：每条事实是原子的、可独立验证的、带置信度的。
    """
    __tablename__ = "profile_facts"

    id: Mapped[str] = mapped_column(String, primary_key=True, index=True)
    user_id: Mapped[str] = mapped_column(String, ForeignKey("users.id"), index=True, nullable=False)
    
    # 画像维度分类
    category: Mapped[str] = mapped_column(String, nullable=False)
    # 可选值: identity, preference, expertise, behavior, goal, 
    #         communication_style, emotional_state, context, custom
    
    # 事实的键值对
    fact_key: Mapped[str] = mapped_column(String, nullable=False)
    # 示例: "programming_language", "preferred_response_style", "active_timezone"
    fact_value: Mapped[str] = mapped_column(Text, nullable=False)
    # 示例: "Python", "concise_and_technical", "UTC+8"
    
    # 置信度与生命周期
    confidence: Mapped[float] = mapped_column(Float, default=0.5)
    # 0.0-1.0，1.0 表示用户明确确认的事实
    source_type: Mapped[str] = mapped_column(String, default="inferred")
    # explicit(用户明确声明), inferred(LLM推断), behavioral(行为分析), 
    # feedback(从反馈学习), manual(手动添加)
    
    # 时间戳
    first_observed_at: Mapped[datetime] = mapped_column(DateTime, nullable=False)
    last_updated_at: Mapped[datetime] = mapped_column(DateTime, nullable=False)
    last_accessed_at: Mapped[datetime] = mapped_column(DateTime, nullable=True)
    
    # 访问与验证
    access_count: Mapped[int] = mapped_column(Integer, default=0)
    verification_count: Mapped[int] = mapped_column(Integer, default=0)
    # 用户确认/否定次数
    
    # 来源引用
    source_session_id: Mapped[Optional[str]] = mapped_column(String(100), nullable=True)
    source_message_ids: Mapped[Optional[List[str]]] = mapped_column(JSON, nullable=True)
    # 可追溯到具体的对话轮次
    
    # 元数据
    fact_metadata: Mapped[Optional[Dict[str, Any]]] = mapped_column(JSON, nullable=True)
    # 扩展字段: { "decay_rate": 0.01, "tags": [...], "language": "zh" }
    
    # 状态
    is_active: Mapped[bool] = mapped_column(Boolean, default=True)
    # False 表示已衰减至无效或被用户删除

    __table_args__ = (
        Index("ix_profile_facts_user_category", "user_id", "category"),
        Index("ix_profile_facts_user_active", "user_id", "is_active"),
        Index("ix_profile_facts_user_confidence", "user_id", "confidence"),
    )
```

### 2.3 置信度与衰减模型

```python
# backend/plugins/user_profile/confidence_model.py

import math
from datetime import datetime, timezone

class ConfidenceModel:
    """
    画像事实的置信度计算与衰减模型。
    
    参考: ACM CHI '24 — "My agent understands me better"
    简化版指数衰减 + 巩固加成。
    """
    
    # 基础衰减率 (每天)
    DEFAULT_DECAY_RATE = 0.01        # ~69 天半衰期
    EMOTIONAL_DECAY_RATE = 0.05      # 情感状态 ~14 天半衰期
    BEHAVIORAL_DECAY_RATE = 0.02     # 行为模式 ~35 天半衰期
    
    # 置信度阈值
    HIGH_CONFIDENCE_THRESHOLD = 0.75
    MEDIUM_CONFIDENCE_THRESHOLD = 0.4
    ARCHIVE_THRESHOLD = 0.15          # 低于此值标记为非活跃
    
    # 来源类型的基础置信度
    SOURCE_BASE_CONFIDENCE = {
        "explicit": 0.95,     # 用户明确声明
        "manual": 0.90,        # 手动添加
        "feedback": 0.80,      # 从用户反馈学习
        "inferred": 0.50,      # LLM 推断
        "behavioral": 0.55,    # 行为分析
    }
    
    @staticmethod
    def calculate_effective_confidence(
        base_confidence: float,
        source_type: str,
        last_updated_at: datetime,
        access_count: int,
        verification_count: int,
    ) -> float:
        """
        计算画像事实的有效置信度。
        
        Args:
            base_confidence: 存储的基础置信度
            source_type: 来源类型
            last_updated_at: 最后更新时间
            access_count: 访问次数
            verification_count: 用户验证次数
            
        Returns:
            0.0-1.0 的有效置信度
        """
        # 计算距今天数
        now = datetime.now(timezone.utc)
        days_since_update = (now - last_updated_at).days
        
        # 来源特定的衰减率
        decay_rate = ConfidenceModel._get_decay_rate(source_type)
        
        # 巩固因子：每次访问/验证降低有效衰减率
        consolidation = 1.0 + 0.1 * math.log(1 + access_count + verification_count * 2)
        effective_decay = decay_rate / consolidation
        
        # 时间衰减
        decay_factor = math.exp(-effective_decay * days_since_update)
        
        # 验证加成
        verification_bonus = min(0.15, verification_count * 0.03)
        
        effective = base_confidence * decay_factor + verification_bonus
        return min(1.0, max(0.0, effective))
    
    @staticmethod
    def _get_decay_rate(source_type: str) -> float:
        if source_type in ("emotional_state",):
            return ConfidenceModel.EMOTIONAL_DECAY_RATE
        elif source_type in ("behavior", "behavioral"):
            return ConfidenceModel.BEHAVIORAL_DECAY_RATE
        return ConfidenceModel.DEFAULT_DECAY_RATE
    
    @staticmethod
    def should_extract(
        last_extraction_at: Optional[datetime],
        conversation_turns_since: int,
    ) -> bool:
        """判断是否应触发新一轮画像提取"""
        if last_extraction_at is None:
            return True
        now = datetime.now(timezone.utc)
        hours_since = (now - last_extraction_at).total_seconds() / 3600
        return hours_since >= 0.5 or conversation_turns_since >= 10
```

### 3.2 画像提取引擎 (`profile_extractor.py`)

核心流程：

```
对话历史 + 行为日志 + 现有画像
    ↓
Prompt 构建（结构化 Schema + Few-shot 示例）
    ↓
LLM 调用（使用项目已有的 model_service）
    ↓
结构化输出解析（提取新的 ProfileFact 列表）
    ↓
与现有画像比对（向量相似度 + 键值匹配）
    ↓
LLM 决策: ADD / UPDATE / DELETE / UNCHANGED（参考 Mem0 模式）
    ↓
写入 ProfileFact 表 + 更新向量存储
    ↓
记录 ExtractionLog
```

关键设计决策：
- **离线异步执行**：使用 `scheduled_task_manager` 或后台任务队列，不阻塞对话主链路
- **非破坏性更新**：不覆盖现有事实，而是通过比对决策生成变更列表
- **结构化输出**：使用 JSON Schema 约束 LLM 输出格式，确保可解析性
- **增量提取**：每次只处理新增的对话内容，节省 token 成本

### 3.5 与 Agent 对话的集成点

```python
# backend/plugins/user_profile_builtin/profile_injector.py

class ProfileInjector:
    """
    将用户画像注入到 Agent 的对话上下文中。
    
    参考 ChatGPT 四层架构，画像放在系统指令之后、对话历史之前。
    采用"零检索"策略：精选画像直接注入，不走 RAG。
    """
    
    MAX_PROFILE_TOKENS = 2000          # 画像最大 token 占用
    MAX_FACTS_PER_CATEGORY = 15        # 每个类别最多保留的事实数
    
    def build_profile_context(self, user_id: str) -> str:
        """
        构建注入 Agent 系统上下文的画像文本。
        
        优先级排序:
        1. identity (身份) — 始终包含
        2. preference + communication_style — 直接影响回复风格
        3. expertise — 影响技术深度
        4. goal — 影响任务理解
        5. behavior + context + emotional_state — 按需包含
        """
        # 获取活跃事实，按有效置信度排序
        facts = self._get_active_facts(user_id)
        
        # 按类别和优先级组织
        blocks = self._organize_into_blocks(facts)
        
        # 编译为 XML 标签格式（参考 Letta Block 编译模式）
        return self._compile_to_context(blocks)
    
    def _compile_to_context(self, blocks: dict) -> str:
        """编译为 Agent 上下文 XML"""
        context_parts = ["<user_profile>"]
        for category, facts in blocks.items():
            if not facts:
                continue
            context_parts.append(f'  <{category}>')
            for fact in facts:
                context_parts.append(f'    <fact key="{fact.fact_key}" '
                                    f'confidence="{fact.confidence:.2f}">'
                                    f'{fact.fact_value}</fact>')
            context_parts.append(f'  </{category}>')
        context_parts.append('</user_profile>')
        return '\n'.join(context_parts)
```

每个画像事实卡片的功能：
- **查看**：事实值、置信度、来源类型、来源追溯（跳转到具体对话）
- **编辑**：用户手动修改值（source_type → "manual"，confidence → 0.90）
- **确认/否定**：✓ 提高置信度并计数验证；✗ 降低置信度并触发重提取
- **删除**：软删除（is_active=False），可恢复
- **拖拽排序**：调整同类别内事实的优先级（影响上下文注入顺序）

## 五、实施步骤（分阶段）

## 六、风险与缓解

| 风险 | 缓解措施 |
|------|---------|
| LLM 提取质量不稳定 | 置信度阈值 + 用户审核机制 + 结构化输出约束 |
| 提取 token 成本过高 | 增量提取（仅新对话） + 定时批量处理 + 小模型（如 GPT-4o-mini） |
| 前端性能（大量事实渲染） | 分页 + 虚拟滚动 + 类别折叠 |
| 与现有系统冲突 | 渐进式替换：新系统先并行运行，验证后逐步迁移 |
| 隐私合规 | 用户完全控制（查看/编辑/删除/导出/清空）+ 数据本地存储 |

## 七、成功指标

| 指标 | 目标 | 测量方式 |
|------|------|---------|
| 画像提取准确率 | >70%（用户确认率） | verification_count / total_facts |
| 提取延迟 | <3秒（不阻塞对话） | 后台异步，不感知 |
| 画像覆盖率 | >5个维度/用户（使用1周后） | categories with ≥3 facts |
| 用户编辑参与率 | >30%（使用1月后至少编辑1次） | facts with source_type="manual" |
| 上下文注入token | ≤2000 tokens | profile_context length |
